# BMI270 Tilt-Nav (Left/Right) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add BMI270 tilt-to-Left/Right input to the M5Stick S3 Flipper port so rolling the stick navigates horizontally, retiring the two-button limitation.

**Architecture:** A pure, host-tested tilt evaluator (`tilt_nav.c`) converts accel samples into Left/Right key events with deadzone + hysteresis + hold-repeat. A minimal accel-only BMI270 driver (`bmi270.c`) feeds it from the internal I2C bus. Both are wired into the existing `target_input_poll()`; a persisted System-settings toggle gates it.

**Tech Stack:** C, ESP-IDF 5.4.1, ESP32-S3, legacy I2C driver (`driver/i2c.h`) on `I2C_NUM_0`, NVS for the toggle. Host tests compiled with `cc`/`gcc`.

## Global Constraints

- Board: `m5stick_s3` only. Guard board-specific code so other boards are unaffected.
- Internal I2C bus is `I2C_NUM_0`, SDA=47, SCL=48, 400 kHz — **already installed** at boot by the M5PM1 PMIC init. Do NOT call `i2c_driver_install` for it; reuse it.
- BMI270 I2C address `0x68` (default). CHIP_ID (reg 0x00) must read `0x24`.
- Tilt tuning start values: FIRE 35 deg, RELEASE 25 deg, REPEAT 150 ms. Left/Right sign and roll/up axis choice are on-device-tunable named constants.
- Absent/failed BMI270 must NOT break buttons — tilt path is skipped silently.
- Left/Right only. Do not alter existing Down/Ok/Up/Back button behavior.
- **This source tree is NOT a git repo.** Where a step says "Checkpoint", it means: run the stated verification and confirm the expected result before moving on. There is no `git commit`. (The webflasher repo under `webflash/` is separate and unrelated to these files.)
- Build command for the firmware: `cmd /c "<abs>\build_only_public.bat"` then confirm `SENTINEL_OK` in `build_only_public.log` and no `error:` lines. (This regenerates sdkconfig from the m5stick defaults — see the ext-manifest notes.)

```c
/* ~+/-2g range: 1g ~= 16384 LSB. Upright => a_up ~ +1g, a_roll ~ 0. */
#define G 16384

static int16_t rroll(double deg) { return (int16_t)(G * sin(deg * 3.14159265 / 180.0)); }

    /* upright -> no key, not engaged */
    assert(tilt_eval(0, G, 0, &st) == TiltNone);
    assert(!st.engaged);

    assert(tilt_eval(rroll(30), rup(30), 10, &st) == TiltNone);
    assert(!st.engaged);

    /* 45 deg right crosses FIRE -> emit Right once, now engaged */
    assert(tilt_eval(rroll(45), rup(45), 20, &st) == TiltRight);
    assert(st.engaged && st.dir == TiltRight);

    /* immediately again (only 5ms later) -> no repeat yet (REPEAT=150) */
    assert(tilt_eval(rroll(45), rup(45), 25, &st) == TiltNone);

    /* 160ms after first emit -> repeat Right */
    assert(tilt_eval(rroll(45), rup(45), 180, &st) == TiltRight);

    /* drop to 30 deg: still >= RELEASE(25) -> stays engaged, no key between repeats */
    assert(tilt_eval(rroll(30), rup(30), 200, &st) == TiltNone);
    assert(st.engaged);

    assert(tilt_eval(rroll(20), rup(20), 210, &st) == TiltNone);
```

```c
#pragma once
#include <stdint.h>
#include <stdbool.h>

/* Tilt-to-direction mapping for the 2-button M5Stick S3. Pure logic, host-testable. */

typedef enum { TiltNone = 0, TiltLeft = 1, TiltRight = 2 } TiltDir;

typedef struct {
    bool engaged;          /* currently past the FIRE threshold in `dir` */
    TiltDir dir;           /* the direction we engaged in */
    uint32_t last_repeat_at; /* ms tick of the last emitted key */
} TiltState;

/* Tuning (on-device tunable). Angles in degrees from upright. */
#define TILT_FIRE_DEG      35.0f
#define TILT_RELEASE_DEG   25.0f   /* hysteresis: must fall below this to disengage */
#define TILT_REPEAT_MS     150u

/* Set to 1 to swap which roll direction is Left vs Right (on-device fix, one line). */
#define TILT_INVERT        0

/* Signed roll angle in degrees. a_up reads ~+/-1g upright; a_roll swings with L/R roll.
 * Uses |a_up| so an upright reading is ~0 deg regardless of the up-axis sign. */
float tilt_roll_deg(int16_t a_roll, int16_t a_up);

/* Evaluate one accel sample. Returns the key to emit THIS tick, or TiltNone.
 * All state lives in *st; `now_ms` is a millisecond clock. */
TiltDir tilt_eval(int16_t a_roll, int16_t a_up, uint32_t now_ms, TiltState* st);
```

- [ ] **Step 4: Write `tilt_nav.c`**

Create `targets/m5stick_s3/tilt_nav.c`:

```c
#include "tilt_nav.h"
#include <math.h>

float tilt_roll_deg(int16_t a_roll, int16_t a_up) {
    /* fabs on the up axis so upright (a_roll==0) is 0 deg for either mounting sign. */
    return atan2f((float)a_roll, fabsf((float)a_up)) * 57.2957795f;
}

TiltDir tilt_eval(int16_t a_roll, int16_t a_up, uint32_t now_ms, TiltState* st) {
    float roll = tilt_roll_deg(a_roll, a_up);
    float mag = roll < 0.0f ? -roll : roll;
    TiltDir cur = (roll >= 0.0f) ? TiltRight : TiltLeft;
#if TILT_INVERT
    cur = (cur == TiltRight) ? TiltLeft : TiltRight;
#endif

    if(!st->engaged) {
        if(mag >= TILT_FIRE_DEG) {
            st->engaged = true;
            st->dir = cur;
            st->last_repeat_at = now_ms;
            return cur; /* first emit */
        }
        return TiltNone;
    }

    /* engaged: released or flipped -> disengage (a later tick may re-fire) */
    if(mag < TILT_RELEASE_DEG || cur != st->dir) {
        st->engaged = false;
        return TiltNone;
    }

    /* held past release in same dir -> repeat on cadence */
    if((uint32_t)(now_ms - st->last_repeat_at) >= TILT_REPEAT_MS) {
        st->last_repeat_at = now_ms;
        return st->dir;
    }
    return TiltNone;
}
```

```c
#include "bmi270.h"
#include "bmi270_config.h"
#include "boards/board.h"

#include <driver/i2c.h>
#include <freertos/FreeRTOS.h>
#include <freertos/task.h>
#include <esp_log.h>
#include <esp_rom_sys.h>

#define TAG "BMI270"
#define BMI270_ADDR        0x68
#define BMI270_I2C_PORT    KB_I2C_PORT   /* I2C_NUM_0 on this board */

/* Registers */
#define REG_CHIP_ID        0x00
#define REG_ACC_DATA       0x0C  /* X_LSB..Z_MSB, 6 bytes */
#define REG_INTERNAL_STATUS 0x21
#define REG_ACC_CONF       0x40
#define REG_ACC_RANGE      0x41
#define REG_INIT_CTRL      0x59
#define REG_INIT_ADDR_0    0x5B
#define REG_INIT_ADDR_1    0x5C
#define REG_INIT_DATA      0x5E
#define REG_PWR_CONF       0x7C
#define REG_PWR_CTRL       0x7D
#define REG_CMD            0x7E

#define BMI270_CHIP_ID     0x24
#define CMD_SOFTRESET      0xB6

static bool s_present = false;

static esp_err_t wr(uint8_t reg, uint8_t val) {
    uint8_t b[2] = {reg, val};
    return i2c_master_write_to_device(BMI270_I2C_PORT, BMI270_ADDR, b, 2, pdMS_TO_TICKS(50));
}
static esp_err_t rd(uint8_t reg, uint8_t* buf, size_t n) {
    return i2c_master_write_read_device(
        BMI270_I2C_PORT, BMI270_ADDR, &reg, 1, buf, n, pdMS_TO_TICKS(50));
}

bool bmi270_is_present(void) { return s_present; }

bool bmi270_init(void) {
    s_present = false;
    uint8_t id = 0;
    if(rd(REG_CHIP_ID, &id, 1) != ESP_OK || id != BMI270_CHIP_ID) {
        ESP_LOGW(TAG, "no BMI270 (CHIP_ID=0x%02X, want 0x24)", id);
        return false;
    }

    wr(REG_CMD, CMD_SOFTRESET);
    vTaskDelay(pdMS_TO_TICKS(5));

    /* disable advanced power save, then upload config (BMI270 needs the blob) */
    wr(REG_PWR_CONF, 0x00);
    esp_rom_delay_us(450);
    wr(REG_INIT_CTRL, 0x00);
    wr(REG_INIT_ADDR_0, 0x00);
    wr(REG_INIT_ADDR_1, 0x00);

    /* burst-write the 8 KB blob to INIT_DATA in chunks */
    const size_t CH = 32;
    uint8_t buf[1 + 32];
    buf[0] = REG_INIT_DATA;
    for(size_t off = 0; off < BMI270_CONFIG_SIZE; off += CH) {
        size_t n = (BMI270_CONFIG_SIZE - off < CH) ? (BMI270_CONFIG_SIZE - off) : CH;
        /* advance the internal INIT pointer (addr in words = off/2) */
        wr(REG_INIT_ADDR_0, (uint8_t)((off / 2) & 0x0F));
        wr(REG_INIT_ADDR_1, (uint8_t)((off / 2) >> 4));
        for(size_t i = 0; i < n; i++) buf[1 + i] = bmi270_config_file[off + i];
        if(i2c_master_write_to_device(BMI270_I2C_PORT, BMI270_ADDR, buf, 1 + n,
                                      pdMS_TO_TICKS(50)) != ESP_OK) {
            ESP_LOGE(TAG, "config upload failed at %u", (unsigned)off);
            return false;
        }
    }
    wr(REG_INIT_CTRL, 0x01);

    /* wait for init_ok */
    bool ok = false;
    for(int i = 0; i < 20; i++) {
        vTaskDelay(pdMS_TO_TICKS(10));
        uint8_t st = 0;
        if(rd(REG_INTERNAL_STATUS, &st, 1) == ESP_OK && (st & 0x0F) == 0x01) { ok = true; break; }
    }
    if(!ok) { ESP_LOGE(TAG, "init_ok never set"); return false; }

    /* enable accel: 100Hz normal filter, +/-2g */
    wr(REG_PWR_CTRL, 0x04);
    wr(REG_ACC_CONF, 0xA8);
    wr(REG_ACC_RANGE, 0x00);
    vTaskDelay(pdMS_TO_TICKS(5));

    s_present = true;
    ESP_LOGI(TAG, "BMI270 ready (accel 100Hz +/-2g)");
    return true;
}

bool bmi270_read_accel(int16_t* x, int16_t* y, int16_t* z) {
    if(!s_present) return false;
    uint8_t d[6];
    if(rd(REG_ACC_DATA, d, 6) != ESP_OK) return false;
    *x = (int16_t)((d[1] << 8) | d[0]);
    *y = (int16_t)((d[3] << 8) | d[2]);
    *z = (int16_t)((d[5] << 8) | d[4]);
    return true;
}
```

```c
static uint8_t system_settings_tilt_load(void) {
    nvs_handle_t h;
    uint8_t v = 1; /* default ON */
    if(nvs_open("input", NVS_READONLY, &h) == ESP_OK) {
```

## Notes for the implementer
- The tilt math is the only non-trivial logic and it is fully covered by the Task 1 host
  test — run that test after any change to `tilt_nav.c`/`.h`.
- Everything else is hardware glue verified by "does it compile" + the Task 5 checklist.
- Keep all new code guarded to / located under `targets/m5stick_s3/` so other boards are
  untouched. `system_settings.c` is shared across boards — the NVS key is harmless on
  boards without a BMI270 (the setter is a no-op there if you also guard
  `target_input_set_tilt_enabled` per board, or simply leave it; other boards have their
  own `target_input.c` and would need their own setter — add a weak/no-op if the linker
  complains).
